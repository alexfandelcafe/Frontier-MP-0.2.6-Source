-- RDRMP Bounty Hunter Tester - client side

-- Flow:
--   /bounty_start [actor enum]
--       -> spawn target near player
--       -> target blip + sheriff blip
--       -> target must actually be lassoed/hogtied
--       -> return target alive to Armadillo Sheriff
--       -> alive = full reward, dead = half reward

-- Tester economy is local to the client and resets on resource restart.

local TARGET_ENUM = 202 -- ACTOR_CAUCASIAN_MALE_TownFolk02
local TARGET_NAME = "rdrmp_bounty_target"

local FULL_REWARD = 1000
local DEAD_REWARD = 500

-- Armadillo Sheriff area used by RDRMP's stock freeroam resource.
local SHERIFF_POS = vector3(-2175.0, 16.3, 2600.5)
local SHERIFF_RADIUS = 45.0

local SPAWN_DISTANCE = 15.0
local SPAWN_SIDE_OFFSET = 3.0

local bounty_actor = 0
local bounty_blip = 0
local sheriff_blip = 0
local bounty_enum = nil

local bounty_active = false
local bounty_captured = false
local bounty_finished = false
local bounty_balance = 0

local function valid_actor(actor)
    return actor ~= nil and actor ~= 0 and natives.entity.is_actor_valid(actor)
end

local function local_message(text)
    print("[rdrmp_bounty] " .. tostring(text))

-- Use the game's own HUD text rather than depending on a server callback.
natives.hud.print_small_b(tostring(text), 5000, true, 0, 0, 0, 0)

end

local function remove_blips()
    if sheriff_blip ~= 0 and natives.hud.is_blip_valid(sheriff_blip) then
        natives.hud.remove_blip(sheriff_blip)
    end

if bounty_blip ~= 0 and natives.hud.is_blip_valid(bounty_blip) then
    natives.hud.remove_blip(bounty_blip)
end

sheriff_blip = 0
bounty_blip = 0

end

local function destroy_bounty_actor()
    if valid_actor(bounty_actor) then
        natives.object.destroy_actor(bounty_actor)
    end

if bounty_enum ~= nil then
    natives.stream.streaming_evict_actor(bounty_enum, -1)
end

bounty_actor = 0
bounty_enum = nil

end

local function clear_bounty_local()
    remove_blips()
    destroy_bounty_actor()
    bounty_active = false
    bounty_captured = false
    bounty_finished = false
end

local function create_target_blip(actor)
    if not valid_actor(actor) then
        return
    end

if bounty_blip ~= 0 and natives.hud.is_blip_valid(bounty_blip) then
    natives.hud.remove_blip(bounty_blip)
end

bounty_blip = natives.hud.add_blip_for_actor(actor, 356, 0.0, 0, 0)

natives.extended.ui_add_string("RDRMP_BOUNTY_TARGET", "Bounty Target")
natives.hud.set_blip_name(bounty_blip, "RDRMP_BOUNTY_TARGET")
natives.hud.set_blip_scale(bounty_blip, 0.9)

end

local function create_sheriff_blip()
    if sheriff_blip ~= 0 and natives.hud.is_blip_valid(sheriff_blip) then
        natives.hud.remove_blip(sheriff_blip)
    end

sheriff_blip = natives.hud.add_blip_for_coord(SHERIFF_POS, 396, 0.0, 2, 0)

natives.extended.ui_add_string("RDRMP_BOUNTY_SHERIFF", "Armadillo Sheriff")
natives.hud.set_blip_name(sheriff_blip, "RDRMP_BOUNTY_SHERIFF")
natives.hud.set_blip_scale(sheriff_blip, 0.9)

end

local function spawn_position_from_player(player_actor)
    local player_pos = natives.object.get_object_position(player_actor)
    local player_heading = natives.object.get_object_heading(player_actor)
    local radians = player_heading * (math.pi / 180.0)

local forward = vector3(
    -math.cos(radians),
    0.0,
    math.sin(radians)
)

local right = vector3(
    math.sin(radians),
    0.0,
    math.cos(radians)
)

return player_pos + (forward * SPAWN_DISTANCE) + (right * SPAWN_SIDE_OFFSET)

end

local function spawn_target(actor_enum)
    actor_enum = tonumber(actor_enum) or TARGET_ENUM

clear_bounty_local()

local player_actor = natives.actor.get_player_actor(-1)
if not valid_actor(player_actor) then
    local_message("El player actor todavía no es válido.")
    return false
end

local enum_name = natives.object.get_actor_enum_string_from_enum(actor_enum)
local_message("Generando objetivo " .. tostring(actor_enum) .. " (" .. tostring(enum_name) .. ").")

if not natives.object.is_actorenum_installed(actor_enum) then
    local_message("El actor enum no está instalado: " .. tostring(actor_enum))
    return false
end

natives.stream.streaming_request_actor(actor_enum, true, false)

local timeout = 10000
local elapsed = 0

while not natives.stream.streaming_is_actor_loaded(actor_enum, -1) and elapsed < timeout do
    thread.wait(0)
    elapsed = elapsed + 1
end

if not natives.stream.streaming_is_actor_loaded(actor_enum, -1) then
    local_message("El asset del objetivo no terminó de cargar.")
    return false
end

local position = spawn_position_from_player(player_actor)
local heading = natives.object.get_object_heading(player_actor)
local rotation = vector3(0.0, heading + 180.0, 0.0)

local layout = natives.object.get_ambient_layout()
bounty_actor = natives.object.create_actor_in_layout(
    layout,
    TARGET_NAME,
    actor_enum,
    position,
    rotation
)

bounty_enum = actor_enum

if not valid_actor(bounty_actor) then
    local_message("create_actor_in_layout devolvió un actor inválido.")
    clear_bounty_local()
    return false
end

-- Target must remain killable so the dead payout can be tested.
natives.tasks.task_stand_still(bounty_actor, -1.0, 0, 0)

create_target_blip(bounty_actor)
create_sheriff_blip()

bounty_active = true
bounty_captured = false
bounty_finished = false

local_message("Objetivo creado cerca del jugador.")
local_message("Capturalo con el LASSO. La proximidad sola NO cuenta.")
local_message("Llevalo al Sheriff de Armadillo.")
local_message("Vivo: $" .. tostring(FULL_REWARD) .. " | Muerto: $" .. tostring(DEAD_REWARD))

return true

end

local function target_is_lassoed_by_player(player_actor, target_actor)
    if not valid_actor(player_actor) or not valid_actor(target_actor) then
        return false
    end

local lasso_target = natives.lasso.get_lasso_target(player_actor)
if lasso_target == target_actor then
    return true
end

local lasso_user = natives.lasso.get_lasso_user(target_actor)
if lasso_user == player_actor then
    return true
end

local hogtie_master = natives.lasso.get_hogtied_master(target_actor)
if hogtie_master == player_actor then
    return true
end

return false

end

local function target_and_player_at_sheriff(player_actor, target_actor)
    if not valid_actor(player_actor) or not valid_actor(target_actor) then
        return false
    end

local player_pos = natives.object.get_object_position(player_actor)
local target_pos = natives.object.get_object_position(target_actor)

return vector3.distance(player_pos, SHERIFF_POS) <= SHERIFF_RADIUS
   and vector3.distance(target_pos, SHERIFF_POS) <= SHERIFF_RADIUS

end

local function finish_bounty(outcome)
    if not bounty_active or bounty_finished then
        return
    end

bounty_finished = true
bounty_active = false

local amount = 0

if outcome == "alive" then
    if not bounty_captured then
        bounty_finished = false
        bounty_active = true
        return
    end

    amount = FULL_REWARD
elseif outcome == "dead" then
    amount = DEAD_REWARD
else
    bounty_finished = false
    bounty_active = true
    return
end

bounty_balance = bounty_balance + amount

if outcome == "alive" then
    local_message("ENTREGA COMPLETADA: +$" .. tostring(amount) .. ".")
else
    local_message("OBJETIVO MUERTO: +$" .. tostring(amount) .. " (50%).")
end

local_message("Saldo tester: $" .. tostring(bounty_balance))

if outcome == "dead" then
    thread.create(function()
        thread.wait(3000)
        clear_bounty_local()
    end)
else
    clear_bounty_local()
end

end

-- Server command bridge events. These handlers do not rely on any server-local
-- mission table, global state or function upvalues.
event.register("rdrmp_bounty:command_start")
event.add_handler("rdrmp_bounty:command_start", function(_args)
    spawn_target(_args)
end)

event.register("rdrmp_bounty:command_clear")
event.add_handler("rdrmp_bounty:command_clear", function()
    clear_bounty_local()
    local_message("Cazarrecompensas cancelado.")
end)

event.register("rdrmp_bounty:command_balance")
event.add_handler("rdrmp_bounty:command_balance", function()
    local_message("Saldo tester de cazarrecompensas: $" .. tostring(bounty_balance))
end)

event.register("core:on_resource_stop")
event.add_handler("core:on_resource_stop", function(name)
    if name == CURRENT_RESOURCE_NAME then
        clear_bounty_local()
    end
end)

thread.create(function()
    while true do
        if bounty_active and valid_actor(bounty_actor) and not bounty_finished then
            local player_actor = natives.actor.get_player_actor(-1)

if valid_actor(player_actor) then
            -- Dead bounty: immediate half reward, no lasso requirement.
            if natives.health.is_actor_dead(bounty_actor) then
                finish_bounty("dead")
            else
                -- Alive bounty: first prove the target was actually lassoed.
                if not bounty_captured and target_is_lassoed_by_player(player_actor, bounty_actor) then
                    bounty_captured = true
                    local_message("Objetivo capturado con LASSO.")
                end

                -- Alive payout requires both player and target inside the sheriff area.
                if bounty_captured and target_and_player_at_sheriff(player_actor, bounty_actor) then
                    finish_bounty("alive")
                end
            end
        end
    end

    thread.wait(0)
end

end)
